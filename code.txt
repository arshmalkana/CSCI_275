        'message': "No Function Found!"
    };
    try {
        if (e.parameter.functions == "login") {
            output = login(e.parameter.orgid.toLowerCase(), e.parameter.password, e.parameter.loginhash);
        } else if (e.parameter.functions == "defaultnewpass") {
            output = defaultnewpass(e.parameter.newpasstoset, e.parameter.loginhash);
        } else if (e.parameter.functions == "checkServerInfo") {
            output = checkServerInfo(e.parameter.loginhash);
        } else if (e.parameter.functions == "informUser") {
            output = informUser(e.parameter.loginhash, e.parameter.orgname.toLowerCase());
        } else if (e.parameter.functions == "updateProfile") {
            output = updateProfile(e.parameter.userName, e.parameter.designation, e.parameter.mobile, e.parameter.email.toLowerCase(), e.parameter.newPass, e.parameter.loginhash);
        } else if (e.parameter.functions == "forgetUser") {
            output = forgetUser(e.parameter.email.toLowerCase());
        } else if (e.parameter.functions == "deleteLastReport") {
            output = deleteLastReport(e.parameter.loginHash);
        } else if (e.parameter.functions == "requestProfile") {
            output = requestProfile(e.parameter.name, e.parameter.institueName, e.parameter.repAuthName, e.parameter.designation, e.parameter.mobile, e.parameter.email.toLowerCase(), e.parameter.userRole, e.parameter.district, e.parameter.tehsil);
        } else if (e.parameter.functions == "createProfile" && e.parameter.key == "asdjhgqweyiuxzcbnaksjdfghporguifdlkghmcxnvjdhfq") {
            output = createProfile(e.parameter.row);
            // } else if (e.parameter.functions == "userPrevMonthlyReports") {
            //     output = userPrevMonthlyReports(e.parameter.loginhash);
        } else if (e.parameter.functions == "mailMonthlyPdf") {
            output = mailMonthlyPdf(e.parameter.loginhash, e.parameter.i);
        } else if (e.parameter.functions == "mailOtherReportData") {
            output = mailOtherReportData(e.parameter.loginhash, e.parameter.reportNo);
        }
    } catch (err) {
      GmailApp.sendEmail("gsmalkana@gmail.com", "Error Log!", "PublicApp's doGet() yielded an error on server: " + err, {
          'from': "arshpreetmalkana@gmail.com",
          'name': "AH Punjab Reporting"
      });

        output = {
            'code': 0,
            'message': "Server Error! Please Report at gsmalkana@gmail.com"
        };
    }

    return ContentService.createTextOutput(JSON.stringify(output));
}

            if (lastReportRowNo != null) {
                SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Form responses").deleteRow(lastReportRowNo);

                updateCredentialsCache();
                var json = {
                    'code': 1,

                var json = {
                    'code': 0,
                    'message': "No Report Found For That Institue!"

            }
        } else {
            var json = {
                'code': 0,
                'message': "Invalid Login!"
            };
        }
    } else {
        var json = {
            'code': 0,
            'message': "Empty Login!"
        };
    }
    return json;
}

function userPrevMonthlyReports(userid) {
    try {
        var data = getFormResponces();
        var rownos = [];
        var dates = [];
        for (var i = 0; i < data.length; i++) {
            if (data[i][4] == userid) {
                rownos.push(i);

                var toDate = new Date(data[i][2].getTime() + (60 * 60 * 24 * 1000));

                dates.push(toDate.toLocaleDateString("en-US", {
                month: 'short'
            }) + "-" + toDate.toLocaleDateString("en-US", {
                year: 'numeric'
            }));
            }
        }

        var json = {
            'code': 1,
            'rownos': rownos,
            'Dates': dates,
        };


        return json;
    } catch (e) {
        return {
            'code': 0,
            'message': "Server Error! Please Report at gsmalkana@gmail.com"
        };
    }
}

      if (((yy % 4) == 0) && (((yy % 100) != 0) || ((yy % 400) == 0))) days[1] = 29;
      for (var i = 0; i < mm; i++) dd += days[i];
      yy--;
      return ((((yy * 365 + ((yy - (yy % 4)) / 4) - ((yy - (yy % 100)) / 100) + ((yy - (yy % 400)) / 400) + dd - 693594) * 24 + hh) * 60 + ii) * 60 + ss) / 86400.0;
  }

 function wait(ms){
   var start = new Date().getTime();
   var end = start;
   while(end < start + ms) {
     end = new Date().getTime();
  }
  }

 function generateString(length) {
    const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
    let result = '';
    const charactersLength = characters.length;
    for (let i = 0; i < length; i++) {

    }

     function exportSheet(spreadsheet, sheetName) {
      var ss = spreadsheet;
      var sheets = ss.getSheets();
      for (var i = 0; i < sheets.length; i++) {
          if (sheetName == sheets[i].getSheetName()) {
              sheets[i].showSheet();
          } else {
              sheets[i].hideSheet();
          }
      }
      ss.getSheetByName(sheetName).showSheet();
      return ss.getBlob();
  }
